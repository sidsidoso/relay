# Relay plan 6b — English/Portuguese UI and end-to-end tests — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The whole web UI (login, inbox, settings, knowledge editor, demo page and widget) is available in English and Brazilian Portuguese with a language switcher, English by default; and Playwright end-to-end tests of the main flows run in CI on the demo provider.

**Architecture:** A dependency-free i18n module: two typed catalogs (`en` is the source of truth, `pt-BR` must have every key, enforced by the type checker), a React context with `t(key, vars)`, the choice stored in `localStorage` (`relay.locale`) and reflected in `<html lang>`. Components read strings through `useI18n()`; without a provider the context falls back to English, so the existing component tests keep working unchanged. Playwright lives in `apps/web/e2e`, starts the real API (demo provider, fresh database in a temp folder, no transcriber) and the Vite dev server, and drives two browser contexts: a visitor on `/demo` and a team member in the inbox.

**Tech Stack:** React 19 + Vite 8, Vitest 5, Testing Library, Playwright 1.65 (Chromium), GitHub Actions.

**Depends on:** plan 6a merged (`.github/workflows/ci.yml`, the `/demo` public notice, the widget "too fast" notice, the `KnowledgeEditor.create` and `Inbox` refresh changes). The component code below already includes those 6a changes.

## Global Constraints

- English is the default UI language; the only other language is Brazilian Portuguese (`pt-BR`). Code, comments, commits and docs stay in English (spec §2).
- What the server writes stays as stored: agent replies, the configured handoff message, handoff reasons and run error codes are not translated by the UI.
- No employer names, data, IPs or keys anywhere; sample data stays fictional (Acme Coworking). CI uses no secrets; e2e runs on the demo provider with transcription off.
- Git identity is already set in the repo. Every commit ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`. Push with `GH_TOKEN=$(gh auth token --user sidsidoso) git push`.
- No new runtime dependency for i18n. The only new dev dependencies are `@playwright/test` and `@types/node` in `apps/web`.
- Run tests per workspace: `npm test -w apps/web -- <file>`; type check with `npm run typecheck`; e2e with `npm run e2e`.

```tsx
import '@testing-library/jest-dom/vitest'
import { render, screen, fireEvent } from '@testing-library/react'
import { LanguageSwitcher, I18nProvider, LOCALE_STORAGE_KEY, storedLocale, translate, useI18n } from '../src/i18n'
import { en } from '../src/locales/en'
import { ptBR } from '../src/locales/pt-BR'

afterEach(() => { localStorage.clear(); document.documentElement.lang = 'en' })

function SignOut() {
  const { t } = useI18n()
  return <button>{t('nav.signOut')}</button>
}

describe('i18n', () => {
  it('translates and fills placeholders', () => {
    expect(translate('en', 'inbox.visitor', { id: 7 })).toBe('Visitor #7')
    expect(translate('pt-BR', 'inbox.visitor', { id: 7 })).toBe('Visitante #7')
    expect(translate('en', 'inbox.handoff', {})).toBe('Handoff: {reason}')
  })

  it('has a non-empty Portuguese text for every English key', () => {
    for (const key of Object.keys(en) as Array<keyof typeof en>) expect(ptBR[key].trim()).not.toBe('')
    expect(Object.keys(ptBR).sort()).toEqual(Object.keys(en).sort())
  })

  it('defaults to English, also without a provider', () => {
    expect(storedLocale()).toBe('en')
    render(<SignOut />)
    expect(screen.getByRole('button', { name: 'Sign out' })).toBeInTheDocument()
  })

  it('switches language, remembers the choice and sets <html lang>', () => {
    render(<I18nProvider><LanguageSwitcher /><SignOut /></I18nProvider>)
    fireEvent.change(screen.getByLabelText('Language'), { target: { value: 'pt-BR' } })
    expect(screen.getByRole('button', { name: 'Sair' })).toBeInTheDocument()
    expect(screen.getByLabelText('Idioma')).toHaveValue('pt-BR')
    expect(localStorage.getItem(LOCALE_STORAGE_KEY)).toBe('pt-BR')
    expect(document.documentElement.lang).toBe('pt-BR')
    expect(storedLocale()).toBe('pt-BR')
  })

  it('ignores an unknown stored value', () => {
    localStorage.setItem(LOCALE_STORAGE_KEY, 'fr')
    expect(storedLocale()).toBe('en')
  })
})
```

`apps/web/src/i18n.tsx`:

```tsx
import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from 'react'
import { en, type MessageKey } from './locales/en'
import { ptBR } from './locales/pt-BR'

export type Locale = 'en' | 'pt-BR'
export type Translate = (key: MessageKey, vars?: Record<string, string | number>) => string

export const LOCALES: Array<{ id: Locale; label: string }> = [
  { id: 'en', label: 'English' },
  { id: 'pt-BR', label: 'Português (Brasil)' },
]
export const LOCALE_STORAGE_KEY = 'relay.locale'

const CATALOGS: Record<Locale, Record<MessageKey, string>> = { en, 'pt-BR': ptBR }

export function translate(locale: Locale, key: MessageKey, vars?: Record<string, string | number>): string {
  const text = CATALOGS[locale][key] ?? en[key]
  if (!vars) return text
  return text.replace(/\{(\w+)\}/g, (match, name: string) => (name in vars ? String(vars[name]) : match))
}

/** The language picked earlier in this browser; English otherwise (spec: English is the default). */
export function storedLocale(): Locale {
  try {
    return localStorage.getItem(LOCALE_STORAGE_KEY) === 'pt-BR' ? 'pt-BR' : 'en'
  } catch {
    return 'en'
  }
}

type I18n = { locale: Locale; setLocale: (l: Locale) => void; t: Translate }

// Without a provider (unit tests of a single component) everything is English.
const I18nContext = createContext<I18n>({ locale: 'en', setLocale: () => {}, t: (k, v) => translate('en', k, v) })

export function I18nProvider({ children, initial }: { children: ReactNode; initial?: Locale }) {
  const [locale, setLocaleState] = useState<Locale>(() => initial ?? storedLocale())
  useEffect(() => { document.documentElement.lang = locale }, [locale])
  const setLocale = useCallback((l: Locale) => {
    setLocaleState(l)
    try { localStorage.setItem(LOCALE_STORAGE_KEY, l) } catch { /* private mode: the choice lasts for this page only */ }
  }, [])
  const t = useCallback<Translate>((k, v) => translate(locale, k, v), [locale])
  const value = useMemo(() => ({ locale, setLocale, t }), [locale, setLocale, t])
  return <I18nContext.Provider value={value}>{children}</I18nContext.Provider>
}

export const useI18n = () => useContext(I18nContext)

export function LanguageSwitcher() {
  const { locale, setLocale, t } = useI18n()
  return (
    // The select is named by aria-label: a wrapping label's text would also include the option texts.
    <label className="lang">
      <span aria-hidden="true">{t('lang.label')}</span>
      <select aria-label={t('lang.label')} value={locale} onChange={(e) => setLocale(e.target.value as Locale)}>
        {LOCALES.map((l) => <option key={l.id} value={l.id}>{l.label}</option>)}
      </select>
    </label>
  )
}
```

```tsx
function Transcript({ t: tr }: { t: Message['transcription'] }) {
  const { t } = useI18n()
  if (!tr) return <span className="msg-transcript muted">{t('transcript.pending')}</span>
  if (tr.status === 'ok') return <p className="msg-transcript"><span className="muted">{t('transcript.label')}</span>{tr.text}</p>
  if (tr.status === 'empty') return <span className="msg-transcript muted">{t('transcript.empty')}</span>
  return <span className="msg-transcript muted">{tr.error ? t('transcript.failedWith', { error: tr.error }) : t('transcript.failed')}</span>
}

export function Timeline({ messages }: { messages: Message[] }) {

    const el = list.current

  }, [messages.length])

    <ol className="timeline" ref={list}>
      {messages.map((m) => (
        <li key={m.id} className={`msg msg-${m.sender}`}>
          <span className="msg-who">{t(`who.${m.sender}`)}</span>

            : <span className="msg-text">{m.text}</span>}
          <time className="msg-time">{new Date(m.createdAt).toLocaleTimeString(locale)}</time>
```

```tsx
import { useCallback, useEffect, useRef, useState } from 'react'
import { api, ApiError, type KnowledgeFile } from '../api'
import { useI18n } from '../i18n'
import type { MessageKey } from '../locales/en'

const NAME = /^[a-z0-9][a-z0-9-]{0,62}\.md$/
const API_ERRORS: Record<string, MessageKey> = {
  invalid_name: 'kb.error.invalid_name',
  too_large: 'kb.error.too_large',
  too_many_files: 'kb.error.too_many_files',
}

type Open = { name: string; content: string; saved: string; isNew: boolean }

export function KnowledgeEditor({ onExpired }: { onExpired: () => void }) {
  const { t } = useI18n()
  const [files, setFiles] = useState<KnowledgeFile[]>([])
  const [open, setOpen] = useState<Open | null>(null)
  const [newName, setNewName] = useState('')
  const [error, setError] = useState<MessageKey | null>(null)
  const [status, setStatus] = useState<'idle' | 'saving' | 'saved'>('idle')
  const busy = useRef(false)

  const fail = useCallback((err: unknown, fallback: MessageKey) => {
    if (err instanceof ApiError && err.status === 401) { onExpired(); return }
    setError(err instanceof ApiError ? API_ERRORS[err.code] ?? fallback : fallback)
  }, [onExpired])

  const refresh = useCallback(async () => {
    try { setFiles((await api.knowledgeList()).files) } catch (err) { fail(err, 'kb.loadFailed') }
  }, [fail])
  useEffect(() => { void refresh() }, [refresh])

  const dirty = open !== null && open.content !== open.saved
  const leave = () => !dirty || window.confirm(t('kb.confirmDiscard'))

  async function select(name: string) {
    if (open?.name === name || !leave()) return
    setError(null); setStatus('idle')
    try {
      const f = await api.knowledgeFile(name)
      setOpen({ name: f.name, content: f.content, saved: f.content, isNew: false })
    } catch (err) { fail(err, 'kb.openFailed') }
  }

  function create() {
    const raw = newName.trim().toLowerCase().endsWith('.md') ? newName.trim() : `${newName.trim()}.md`
    if (!NAME.test(raw)) { setError('kb.error.invalid_name'); return }
    setNewName('')
    // An existing name opens that file: a blank editor would overwrite it on save.
    if (files.some((f) => f.name === raw)) { void select(raw); return }
    if (!leave()) return
    setError(null); setStatus('idle')
    setOpen({ name: raw, content: '', saved: '', isNew: true })
  }

  async function save() {
    if (!open || busy.current) return
    busy.current = true
    setStatus('saving'); setError(null)
    const { name, content } = open
    try {
      await api.saveKnowledge(name, content)
      setOpen((o) => (o && o.name === name ? { ...o, saved: content, isNew: false } : o))
      setStatus('saved')
      await refresh()
    } catch (err) {
      setStatus('idle')
      fail(err, 'kb.saveFailed')
    } finally {
      busy.current = false
    }
  }

  async function remove() {
    if (!open || busy.current || !window.confirm(t('kb.confirmDelete', { name: open.name }))) return
    busy.current = true
    try {
      if (!open.isNew) await api.deleteKnowledge(open.name)
      setOpen(null); setStatus('idle')
      await refresh()
    } catch (err) {
      fail(err, 'kb.deleteFailed')
    } finally {
      busy.current = false
    }
  }

  return (
    <section className="kb" aria-label={t('kb.title')}>
      <h2>{t('kb.title')}</h2>
      <p className="muted">{t('kb.helpBefore')}<code>## heading</code>{t('kb.helpAfter')}</p>
      <div className="kb-grid">
        <ul className="kb-files">
          {files.map((f) => (
            <li key={f.name}>
              <button type="button" aria-pressed={open?.name === f.name} onClick={() => void select(f.name)}>{f.name}</button>
            </li>
          ))}
          {files.length === 0 && <li className="muted">{t('kb.noFiles')}</li>}
        </ul>
        <div className="kb-editor">
          {open ? (
            <>
              <label>
                {t('kb.contentOf', { name: open.name })}
                <textarea rows={18} value={open.content} onChange={(e) => { const v = e.target.value; setOpen((o) => (o ? { ...o, content: v } : o)); setStatus('idle') }} />
              </label>
              <div className="settings-actions">
                <button type="button" onClick={() => void save()} disabled={status === 'saving'}>{t('kb.saveFile')}</button>
                <button type="button" onClick={() => void remove()}>{t('kb.delete')}</button>
                {status === 'saved' && <span className="muted" role="status">{t('kb.saved')}</span>}
                {dirty && <span className="muted">{t('kb.unsaved')}</span>}
              </div>
            </>
          ) : <p className="empty">{t('kb.selectOrCreate')}</p>}
        </div>
      </div>
      <div className="settings-actions">
        <label>{t('kb.newName')} <input value={newName} onChange={(e) => setNewName(e.target.value)} placeholder="parking" /></label>
        <button type="button" onClick={create}>{t('kb.newFile')}</button>
      </div>
      {error && <p role="alert" className="error">{t(error)}</p>}
    </section>
  )
}
```

```ts
import { expect, type Browser, type Page } from '@playwright/test'

/** A visitor: a fresh browser context (so a fresh visitor token) on /demo with the widget open. */
export async function openWidget(browser: Browser, opts: { microphone?: boolean } = {}): Promise<Page> {
  const context = await browser.newContext(opts.microphone ? { permissions: ['microphone'] } : {})
  const page = await context.newPage()
  await page.goto('/demo')
  await page.getByRole('button', { name: 'Chat with us' }).click()
  return page
}

/** Sends a text message; retries while the socket is still connecting (the draft is kept until it is sent). */
export async function sendFromWidget(page: Page, text: string): Promise<void> {
  const input = page.getByRole('textbox', { name: 'Message' })
  await input.fill(text)
  await expect(async () => {
    if ((await input.inputValue()) !== '') await page.getByRole('form', { name: 'Chat' }).getByRole('button', { name: 'Send' }).click()
    await expect(input).toHaveValue('', { timeout: 1000 })
  }).toPass()
  await expect(page.locator('.widget-body')).toContainText(text)
}

/** A team member signed in to the inbox with the pre-filled demo credentials. */
export async function signIn(browser: Browser): Promise<Page> {
  const page = await (await browser.newContext()).newPage()
  await page.goto('/')
  await page.getByRole('button', { name: 'Sign in' }).click()
  await expect(page.getByRole('button', { name: 'Sign out' })).toBeVisible()
  return page
}

/** Opens the most recently updated conversation, optionally under a filter ("Waiting for human"…). */
export async function openNewestConversation(page: Page, filter?: string): Promise<void> {
  if (filter) await page.getByRole('button', { name: filter, exact: true }).click()
  await page.locator('.conv-list .conv').first().click()
}
```
